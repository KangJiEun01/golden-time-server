const params = new URLSearchParams(location.search); const channel = params.get('channel'); const tokenUrl = params.get('tokenUrl'); let client = null; let pendingAudioTrack = null; window.onload = () => joinChannel(); async function joinChannel() { setStatus('🔄 연결 중...'); try { const res = await fetch(`${tokenUrl}/api/token?channel=${channel}&uid=999`); const data = await res.json(); client = AgoraRTC.createClient({ mode:'rtc', codec:'vp8' }); // 이미 채널에 있는 유저 감지 client.on('user-published', async (user, type) => { await client.subscribe(user, type); if (type === 'audio') { pendingAudioTrack = user.audioTrack; setStatus('🔔 소리 수신됨 — 듣기 버튼을 누르세요'); document.getElementById('listenBtn').style.background = '#ff3b30'; } }); client.on('user-unpublished', () => { setStatus('연결이 종료되었습니다.'); document.getElementById('wave').classList.remove('active'); }); await client.join(data.appId, channel, data.token, 999); // 이미 채널에 있는 유저 확인 const remoteUsers = client.remoteUsers; if (remoteUsers.length > 0) { for (const user of remoteUsers) { if (user.hasAudio) { await client.subscribe(user, 'audio'); pendingAudioTrack = user.audioTrack; setStatus('🔔 소리 수신됨 — 듣기 버튼을 누르세요'); document.getElementById('listenBtn').style.background = '#ff3b30'; } } } else { setStatus('✅ 연결됨 — 듣기 버튼을 눌러주세요'); } } catch(e) { setStatus('연결 실패: ' + e.message); } } function startListening() { const btn = document.getElementById('listenBtn'); if (pendingAudioTrack) { pendingAudioTrack.play(); setStatus('🔴 실시간 청취 중'); document.getElementById('wave').classList.add('active'); btn.textContent = '청취 중...'; btn.disabled = true; } else { setStatus('⏳ 아직 소리가 없습니다. 잠시 후 다시 누르세요.'); } } function setStatus(msg) { document.getElementById('status').textContent = msg; }
